# recipes/build_by_hand.txt — Heimdall's stack, built by hand: commands to copy and paste into a
# terminal, one step at a time, in this order. Ubuntu 24.04 (the lab PC's system); the versions and
# flags are packages.toml's (recipes/README.md §3 says why each flag is there).
#
# How to use it:
#   - Paste STEP 0 first, in every new terminal: it sets the variables and the helper every later step
#     uses. Closed the terminal half way? Open a new one, paste STEP 0 again, carry on where you were.
#   - Paste one step at a time, and wait for it to finish. Each ends with a check: if the check prints
#     an error, stop there and read the last lines above it.
#   - The whole file is also valid bash, but pasting step by step lets you see where something fails.
#     (python3 recipes/build_stack.py does all of this by itself, with resume and logs.)
#
# Time: about 3 hours for the core stack on 16 cores (ROOT alone 1-2 h). Disk: about 20 GB.


# ═════════════════════════════════════════════════════════════════════════════════════════════════
# STEP 0 — the variables and a helper (paste in every new terminal)
# ═════════════════════════════════════════════════════════════════════════════════════════════════

export HEP=$HOME/Github/Heimdall                      # Heimdall: everything goes into it
export SRC=$HEP/stash BUILD=$HEP/forge HEP_INSTALL=$HEP/stack VENV=$HEP/env/venv
export CORES=$(nproc)                                 # compilers at once: make -j$CORES
export ROOT_CORES=$(( $(awk '/MemTotal/ {print $2}' /proc/meminfo) / 3000000 ))   # ROOT: one per 3 GB RAM
[ "$ROOT_CORES" -ge 1 ] || ROOT_CORES=1; [ "$ROOT_CORES" -le "$CORES" ] || ROOT_CORES=$CORES
export CXXFLAGS="-O2 -std=c++17"                      # one C++ standard for the whole stack
export PY=$(python3 -c 'import sys; print("%d.%d" % sys.version_info[:2])')      # e.g. 3.12
export PYINC=$(python3 -c 'import sysconfig; print(sysconfig.get_paths()["include"])')
export VIRTUAL_ENV=$VENV
export LHAPDF_DATA_PATH=$HEP_INSTALL/LHAPDF/share/LHAPDF
for d in LHAPDF hepmc3 fastjet root yoda rivet pythia8 herwig7 delphes sherpa whizard geant4; do
    export PATH=$HEP_INSTALL/$d/bin:$PATH
    export LD_LIBRARY_PATH=$HEP_INSTALL/$d/lib${LD_LIBRARY_PATH:+:$LD_LIBRARY_PATH}
    export PYTHONPATH=$HEP_INSTALL/$d/lib/python$PY/site-packages${PYTHONPATH:+:$PYTHONPATH}
done
export PATH=$VENV/bin:$PATH
mkdir -p "$SRC" "$BUILD" "$HEP_INSTALL"

# get NAME URL: download a tarball into $SRC (once) and unpack it, fresh, into $SRC/NAME
get() {
    [ -s "$SRC/$1.tar" ] || wget -O "$SRC/$1.tar" "$2" || { rm -f "$SRC/$1.tar"; return 1; }
    rm -rf "$SRC/$1" && mkdir -p "$SRC/$1" && tar xf "$SRC/$1.tar" -C "$SRC/$1" --strip-components=1
}

echo "HEP=$HEP  cores=$CORES  ROOT cores=$ROOT_CORES  python=$PY"

# ═════════════════════════════════════════════════════════════════════════════════════════════════
# STEP 4 — HepMC3 3.3.1 (its Python module into the venv; -fno-lto keeps the pybind11 symbols)
# ═════════════════════════════════════════════════════════════════════════════════════════════════

get hepmc3 "https://hepmc.web.cern.ch/hepmc/releases/HepMC3-3.3.1.tar.gz"

    -DCMAKE_BUILD_TYPE=Release -DCMAKE_CXX_STANDARD=17 \
    -DHEPMC3_ENABLE_PYTHON=ON -DHEPMC3_PYTHON_VERSIONS="$PY" \
    -DHEPMC3_Python_SITEARCH"${PY/./}"="$VENV/lib/python$PY/site-packages" \
    -DHEPMC3_ENABLE_ROOTIO=OFF -DHEPMC3_ENABLE_SEARCH=ON -DHEPMC3_BUILD_EXAMPLES=OFF \
    -DCMAKE_CXX_FLAGS="$CXXFLAGS -fno-lto" -DCMAKE_SHARED_LINKER_FLAGS=-fno-lto
cmake --build "$BUILD/hepmc3" -j"$CORES" && cmake --install "$BUILD/hepmc3"
# check:
HepMC3-config --version && python -c 'import pyHepMC3; print("pyHepMC3 OK")'

make -j"$CORES" && make install

# check:
